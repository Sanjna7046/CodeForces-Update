
 ## Vectors
v.push_back(x);   // Add x at the end

v.pop_back();     // Remove the last element

v.back();         // Access the last element

v.front();        // Access the first element

v.size();         // Number of elements

v.empty();        // Returns true if the vector is empty

```cpp
#include <bits/stdc++.h>

using namespace std;

int main() {

    int T;
    cin >> T;

    while (T--) {

        int n, k, m;
        cin >> n >> k >> m;
        if (k > m)
            cout << "no";
        else {
            vector < long long > v(n);
            for (int i = 0; i <n; i++) {
                v[i]=i%k==0?m-k+1:1;
            }
           
            cout << "yes" << endl;
            for (auto h: v)
                cout << h << " ";
        }
        cout << endl;
    }
}
```
